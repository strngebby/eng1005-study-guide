## 5. Inverse by Gauss–Jordan

<StepReveal
	problem={"A = [1 1 0]\n    [0 1 2]\n    [1 0 1]"}
	steps={[["R3 − R1, then R3 + R2","[1 1 0 |  1 0 0]\n[0 1 2 |  0 1 0]\n[0 0 3 | −1 1 1]"],["R3 ÷ 3","[0 0 1 | −1/3 1/3 1/3]"],["R2 − 2R3, then R1 − R2","[1 0 0 |  1/3 −1/3  2/3]\n[0 1 0 |  2/3  1/3 −2/3]\n[0 0 1 | −1/3  1/3  1/3]"],["Result","A⁻¹ = ⅓ [1 −1 2; 2 1 −2; −1 1 1]"]]}
/>
